    def get_summary(self, quantiles: Sequence[float] = (0.05, 0.5, 0.95)) -> list[dict[str, object]]:
    """Return statistics of every numeric scalar output per row and scenario.

    They are the mean, the standard deviation, and the quantiles.

    Args:
        quantiles:
            The quantiles to report.

    Returns:
        summary:
            One row per design row and scenario, with `<output>.mean`, `<output>.std`, and `<output>.q<quantile>`
            columns and the number of replicates as `runs`.
    """
    groups: dict[tuple[tuple[tuple[str, object], ...], str], list[_RunResultSpec]] = defaultdict(list)
    for run in self.runs:
        groups[(tuple(run.row.items()), run.scenario)].append(run)
    summary = []
    for (row, scenario), runs in groups.items():
        entry: dict[str, object] = {**dict(row), 'scenario': scenario, 'runs': len(runs)}
        for name in runs[0].scalars:
            values = np.asarray([run.scalars[name] for run in runs])
            if values.ndim != 1 or not np.issubdtype(values.dtype, np.number):
                continue
            values = values.astype(float)
            entry[f'{name}.mean'] = float(values.mean())
            entry[f'{name}.std'] = float(values.std(ddof=1)) if values.size > 1 else 0.0
            for quantile in quantiles:
                entry[f'{name}.q{quantile:g}'] = float(np.quantile(values, quantile))
        summary.append(entry)
    return summary

    basin_stability(system, bounds, classifier, *, n=500, until=None, dt=None, random_state=None, executor='serial', vectorized=False)

    Return the share of random initial states that end in each final behaviour.

    An attractor is the state or the cycle that a run settles into.             The number of chains of MCMC.

            The solver, or `None` for the tuned one.

            `'serial'`, `'thread'`, `'process'`, or a `concurrent.futures` executor that runs the calibrations.

        result:
            The calibration of every individual and the table of their estimates.

    Raises:
        ValidationError:
            There is no individual, a known parameter is also fitted or belongs to an individual without data, or
            `calibrate` rejects a setting.

            The system is a name.

        Two populations grow at their own rates from their own known initial sizes.

        >>> import numpy as np
        >>> from skcomplex.core.experiments import calibrate_individuals
        >>> from skcomplex.systems.biology.ecology import LogisticGrowthSystem
        >>> times = np.arange(0.0, 21.0)
        >>> data = {
        ...     'north': {'population.x': (times, 100 / (1 + 9 * np.exp(-0.4 * times)))},
        ...     'south': {'population.x': (times, 100 / (1 + 4 * np.exp(-0.7 * times)))},
        ... }
        >>> known = {'north': {'n': 10.0}, 'south': {'n': 20.0}}
        >>> result = calibrate_individuals(LogisticGrowthSystem(), data, {'r': (0.1, 2.0)}, known=known)
        >>> result.estimates['r'].round(2).to_dict()
        {'north': 0.4, 'south': 0.7}
    """
    source = _check_source(system)
    if not data:
        error_msg = 'Calibrating individuals needs the data of at least one individual.'
        raise ValidationError(error_msg)
    known = known or {}
    for individual, values in known.items():
        if individual not in data:
            error_msg = f'The individual {individual!r} has known parameters but no data.'
            raise ValidationError(error_msg)
        fitted = sorted(set(values) & set(bounds))
        if fitted:
            error_msg = (
                f'The parameter {", ".join(fitted)} of the individual {individual!r} is both known and fitted. Remove '
                'it from `known` or from `bounds`.'
            )
            raise ValidationError(error_msg)
    sources: list[Source] = []
    for individual in data:
        values = dict(known.get(individual, {}))
        if isinstance(source, BaseSystem):
            sources.append(build_system(source, values))
        else:
            sources.append(functools.partial(source, **values))
    rng = np.random.default_rng(random_state)
    seeds = [None if random_state is None else int(seed) for seed in rng.integers(SEED_RANGE, size=len(data))]
    fits = [
        functools.partial(
            calibrate,
            individual_source,
            individual_data,
            bounds,
            method=method,
            loss=loss,
            initial=initial,
            n_samples=n_samples,
            budget=budget,
            chains=chains,
            replicates=replicates,
            dt=dt,
            solver=solver,
            random_state=seed,
        )
        for individual_source, individual_data, seed in zip(sources, data.values(), seeds, strict=True)
    ]
    calibrations = dict(zip(data, _map_calls(operator.call, fits, executor), strict=True))
    table = pd.DataFrame(
        [result.estimates for result in calibrations.values()],
        index=pd.Index(list(calibrations), name='individual'),
    )
    return IndividualCalibrationResult(calibrations, table)

    Return which model parameters of a system the observed time series determine.

    A calibration returns estimates even when the data cannot pin them down. This function tells practical identifiability, whether the data at hand determine each parameter within its bounds. method says how:

    • 'profile': the profile likelihood with pyPESTO. The best fit is the best of n_starts local fits of the Gaussian likelihood with an unknown variance. The profile of a parameter fixes it at values on both sides of the best fit and fits the others again, until the likelihood ratio falls below the threshold of the level or the parameter reaches its bound. The interval holds the values whose ratio stays above the threshold of the chi-square test with one degree of freedom. A parameter is 'identifiable' when both ends of its interval lie inside its bounds, and 'not identifiable' otherwise.

Each of n copies of the values it simulates at the data times takes Gaussian noise relative to the values, at every level of noise, and is fitted again. The average relative error of a parameter at a level gives the verdict 'strong' when it is at most the level, 'weak' when it is at most ten times the level, and 'none' above. A parameter takes its worst verdict over the levels.

    A profile costs many fits, and refits cost n fits per level, each of many runs of the system.

    random_state: int | None = None,
    executor: str | Executor = 'serial',
) -> IdentifiabilityResult:
    """Return which model parameters of a system the observed time series determine.

    A calibration returns estimates even when the data cannot pin them down. This function tells practical
    identifiability, whether the data at hand determine each parameter within its bounds. `method` says how:

    - `'profile'`: the profile likelihood with pyPESTO. The best fit is the best of `n_starts` local fits of the
      Gaussian likelihood with an unknown variance. The profile of a parameter fixes it at values on both sides of the
      best fit and fits the others again, until the likelihood ratio falls below the threshold of the level or the
      parameter reaches its bound. The interval holds the values whose ratio stays above the threshold of the
      chi-square test with one degree of freedom. A parameter is `'identifiable'` when both ends of its interval lie
      inside its bounds, and `'not identifiable'` otherwise.
    - `'refits'`: the refits under noise of Liyanage et al. (2025). The best fit is a local fit of `calibrate`. Each of
      `n` copies of the values it simulates at the data times takes Gaussian noise relative to the values, at every
      level of `noise`, and is fitted again. The average relative error of a parameter at a level gives the verdict
      `'strong'` when it is at most the level, `'weak'` when it is at most ten times the level, and `'none'` above. A
      parameter takes its worst verdict over the levels.

    A profile costs many fits, and refits cost `n` fits per level, each of many runs of the system.

    It appears in many fields. It tells whether the growth rate and the capacity of a population follow from counts,
    whether the rate of infection and the rate of recovery of an epidemic follow from case counts, and whether the
    clearance of a drug follows from measured concentrations.

    Args:
        system:
            The system, whose copies take the model parameters through `set_params`, or a factory of systems.

            A mapping from state or observable path to its observed `(times, values)`.

            A mapping from model parameter to its `(low, high)` bounds or its SciPy prior, as `calibrate` takes them.

        method:
            `'profile'` or `'refits'`.

            The confidence level of the likelihood-ratio intervals of a profile, between 0 and 1.

            The number of local fits of a profile whose best is the best fit.

        noise:
            The relative noise levels of the refits, such as 0.01 for 1% of every value.

            The number of noisy copies per level of the refits.

        dt:
            The time between frames, or `None` for the tuned one.

        solver:
            The solver, or `None` for the tuned one.

        random_state:
            The seed of the start points, the runs, and the noise.

        executor:
            `'serial'`, `'thread'`, `'process'`, or a `concurrent.futures` executor that runs the refits.

        True

        Data that stop while the population is small say little about the capacity.

        >>> early = np.linspace(0.0, 3.0, 30)
        >>> values = 100 / (1 + 99 * np.exp(-0.5 * early)) * noise
        >>> result = identifiability(
        ...     LogisticGrowthSystem(n=1.0), {'population.x': (early, values)}, bounds, random_state=0

    """

    if method not in METHODS:
        error_msg = f'Unknown method {method!r}. Use one of {", ".join(repr(item) for item in METHODS)}.'
        raise ValidationError(error_msg)

        raise ValidationError(error_msg)
    if not noise or any(not float(item) > 0 for item in noise):
        error_msg = f'The noise levels of the refits must be positive, not {tuple(noise)!r}.'
        raise ValidationError(error_msg)
    _check_count('n', n)
    _check_count('n_starts', n_starts)
    _check_settings('local', 'likelihood', data, bounds, 1)

            source,
            data,
            bounds,
            noise=tuple(float(item) for item in noise),
            n=n,
            dt=dt,
            solver=solver,
            random_state=random_state,
            executor=executor,
        )
    objective = _CalibrationObjective(
        source,
        _series_of(data),
        _check_priors(bounds),
        dt=dt,
        solver=solver,
        random_state=random_state,
    )

    system: Source,
    bounds: Bounds,
    times: npt.ArrayLike,
    *,
    paths: Sequence[str] | None = None,
    method: str = 'mcmc',
    loss: str = 'likelihood',
    n: int = 40,

    It draws `n` true parameter vectors from the priors the bounds give, simulates data at each at the given times with
    Gaussian noise of a standard deviation, calibrates every data set with the posterior method, and counts how often
    the central credible interval of every level holds the true value. It also gives the rank of every true value among
    its samples, the check of simulation-based calibration (Talts et al. 2018).

    It appears in every Bayesian calibration that needs to check its intervals, such as of the rates of an epidemic, of
    the parameters of a climate model, or of the constants of a reaction, before trusting them on real data.

    Args:
        system:
            The system, or a factory of systems.

        bounds:
            A pair `(low, high)` or a SciPy prior per model parameter, as `calibrate` takes them.

        times:
            The times of the simulated data.

        paths:
            The state or observable paths of the data, or `None` for every scalar state.

        method:
            `'abc_smc'`, `'mcmc'`, or `'variational'`.

        loss:
            The loss of the method.

        n:
            The number of true parameter vectors.

        levels:
            The levels of the central credible intervals, between 0 and 1.

        noise:
            The standard deviation of the Gaussian noise added to the simulated data.

        summary:
            The summary statistics of ABC-SMC, or `None`.

        n_samples:
            The number of samples of every calibration, per chain for MCMC.

        budget:
            The budget of every calibration, or `None` for the default of the method.

        chains:
            The number of chains of MCMC.

        dt:
            The time between frames, or `None` for the tuned one.

        solver:
            The solver, or `None` for the tuned one.

        random_state:
            The seed of the true values, the noise, and the calibrations.

        executor:
            `'serial'`, `'thread'`, `'process'`, or a `concurrent.futures` executor that runs the calibrations. MCMC
            and variational calibration run numpyro, whose effect handlers are global, so they do not run on threads.

    Returns:
        result:
            The coverage by level and parameter, the ranks, the true values, and the calibrations.

    Raises:
        ValidationError:
            The method is not a posterior method, the loss does not fit it, a level is not between 0 and 1, `n` is not
            a positive integer, or the executor is unknown or runs numpyro on threads.

    Examples:
        >>> import numpy as np
        >>> from skcomplex.core.experiments import posterior_coverage
        >>> from skcomplex.systems.biology.ecology import LogisticGrowthSystem
        >>> result = posterior_coverage(
        ...     LogisticGrowthSystem(n=10.0),
        ...     {'r': (0.5, 1.0)},
        ...     np.arange(0.0, 11.0),
        ...     method='variational',
        ...     n=3,
        ...     levels=(0.9,),
        ...     noise=5.0,
        ...     n_samples=100,
        ...     budget=300,
        ...     random_state=0,
        ... )
        >>> sorted(result.coverage[0.9]), result.ranks['r'].shape
        (['r'], (3,))
    """
    source = _check_source(system)
    if method not in POSTERIOR_METHODS:
        error_msg = f'Posterior coverage calibrates by a posterior method. Use one of {", ".join(POSTERIOR_METHODS)}.'
        raise ValidationError(error_msg)
    if loss not in LOSSES[method] or loss == 'events':
        allowed = ', '.join(item for item in LOSSES[method] if item != 'events')
        error_msg = f'Posterior coverage with the method {method!r} takes the losses {allowed}.'
        raise ValidationError(error_msg)
    _check_count('n', n)
    checked = tuple(float(level) for level in levels)
    if not checked or any(not 0 < level < 1 for level in checked):
        error_msg = f'The levels of posterior coverage must lie between 0 and 1, not {tuple(levels)!r}.'
        raise ValidationError(error_msg)
    if isinstance(executor, str) and executor not in EXECUTORS:
        error_msg = f"executor = {executor!r} must be 'serial', 'thread', 'process', or an executor."
        raise ValidationError(error_msg)
    if (executor == 'thread' or isinstance(executor, ThreadPoolExecutor)) and method != 'abc_smc':
        error_msg = (
            f'The method {method!r} runs numpyro, whose effect handlers are global, so it cannot run on threads. Use '
            "executor='process' or executor='serial'."
        )
        raise ValidationError(error_msg)
    priors = _check_priors(bounds)
    rng = np.random.default_rng(random_state)
    draws = {name: np.asarray(prior.rvs(size=n, random_state=rng), dtype=float) for name, prior in priors.items()}
    truths = [{name: float(values[index]) for name, values in draws.items()} for index in range(n)]
    seeds = [int(seed) for seed in rng.integers(2**31, size=n)]
    task = functools.partial(
        _calibrate_truth,
        source=source,
        bounds=bounds,
        times=np.asarray(times, dtype=float),
        paths=paths,
        noise=noise,
        method=method,
        loss=loss,
        summary=summary,
        n_samples=n_samples,
        budget=budget,
        chains=chains,
        dt=dt,
        solver=solver,
    )
    results = tuple(_map_calls(task, list(zip(truths, seeds, strict=True)), executor))
    coverage: dict[float, dict[str, float]] = {level: {} for level in checked}
    ranks: dict[str, Array] = {}
    for name in priors:
        pairs = [((result.samples or {})[name], np.asarray(result.weights)) for result in results]
        values = [truth[name] for truth in truths]
        below = [np.sum(weights[samples < value]) for (samples, weights), value in zip(pairs, values, strict=True)]
        ranks[name] = np.asarray(below) / np.asarray([np.sum(weights) for _, weights in pairs])
        for level in checked:
            intervals = [_interval(samples, weights, level) for samples, weights in pairs]
            inside = [low <= value <= high for (low, high), value in zip(intervals, values, strict=True)]
            coverage[level][name] = float(np.mean(inside))
    return CoverageResult(checked, coverage, ranks, draws, results)

        raise ValidationError(error_msg)

    acceptance = len(plausible) / n
    if target is None:
        return VirtualPopulationResult(plausible, acceptance, None, None)
    [(name, observed)] = target.items()
    if name not in kept:
        error_msg = f'The target {name!r} names no output. The outputs are {", ".join(kept)}.'
        raise ValidationError(error_msg)
    if len(plausible) < MIN_PLAUSIBLE:
        error_msg = (
            f'Selection by inclusion estimates densities and needs at least {MIN_PLAUSIBLE} plausible rows, not '
            f'{len(plausible)}. Sample more rows or widen the constraints.'
        )
        raise ExperimentError(error_msg)
    inclusion = _get_inclusion(np.asarray(observed, dtype=float), plausible[name].to_numpy(dtype=float))
    rng = np.random.default_rng(np.random.SeedSequence(random_state).spawn(1)[0])
    selected = plausible[rng.random(len(plausible)) < inclusion].reset_index(drop=True)
    return VirtualPopulationResult(plausible, acceptance, selected, inclusion)

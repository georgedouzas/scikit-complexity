    """Return the advection of a field expression by a velocity, its rate of transport, with upwind differences.

    Args:

    Args:
        angle:
            The angle of one member, in radians.

        over:
            The namespace of the group to reduce over, or `None` for the group the angle reads.

        where:
            A condition on the reduced member that selects the members, or `None` for all members.

        weights:
            A constant matrix with one row per reading member and one column per reduced member, or `None`.

        by:
            The name of a state of the group, so every member reduces over the members that share its value of the
            state, or `None` for all members. It needs the namespace of the group as `over`.

    Returns:
        reduction:
            The symbolic circular mean, in `[-pi, pi]`.

    Raises:
        ValidationError:
            The group is not a namespace or a symbol, the weights are not a matrix, or a key has no group namespace.

    Examples:
        >>> from sage.all__sagemath_symbolics import var
        >>> from skcomplex.core.laws.utils import circular_mean
        >>> circular_mean(var('theta'))
        arctan2(total(sin(theta)), total(cos(theta)))
    """
    sines = total(sin(SR(angle)), over, where, weights, by)
    cosines = total(cos(SR(angle)), over, where, weights, by)
    return arctan2(sines, cosines)

    """Return an expression limited to an interval, `low` below it, `high` above it, and the expression inside.

    Args:
        x:
            The expression to limit.

        low:
            The lower bound, or `None` for none.

    Returns:
        value:
            The symbolic limited value.

        >>> from sage.all__sagemath_symbolics import var
        >>> from skcomplex.core.laws.utils import clip
        >>> x = var('x')
        >>> clip(x, 0, 1)

    if high is not None:
        value = where(SR(x) > high, high, value)
    if low is not None:
        value = where(SR(x) < low, low, value)
    return value

        for _ in range(capacity):
            order = rng.permutation(len(left))
            for first, second in zip(order[::2], order[1::2], strict=False):
                key = (min(first, second), max(first, second))
                if key not in seen:
                    seen.add(key)
                    pairs.append((left[first], right[second]))

    seconds = rng.permutation(np.repeat(np.arange(len(right)), capacity))
    for first, second in zip(firsts, seconds, strict=False):
        if (first, second) not in seen:
            seen.add((first, second))
            pairs.append((left[first], right[second]))
    return pairs

        >>> first.level, second.level
        (6.0, 4.0)
    """
    target = state if sink_state is None else sink_state
    for agent, name in ((source, state), (sink, target)):
        if not hasattr(agent, name):
            error_msg = f'The transfer of {amount} reads the state {name!r}, which the agent {agent!r} does not have.'
            raise ValidationError(error_msg)
    if source is sink and state == target:
        return
    source_value, sink_value = getattr(source, state), getattr(sink, target)
    setattr(source, state, source_value - amount)
    setattr(sink, target, sink_value + amount)

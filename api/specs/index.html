    def values(self, x: Array, p: Array, sites: Indices | None = None) -> dict[str, Array | float]:
    """Return the values of the arguments, per-site values restricted to some sites.

    Args:
        x:
            The state vector.

        p:
            The parameter vector.

        sites:
            The sites to keep, or `None` for all.

    Returns:
        values:
            The value of every argument by name.
    """
    found: dict[str, Array | float] = {}
    for name, is_state, start, size in zip(self.names, self.vectors, self.starts, self.sizes, strict=True):
        source = x if is_state else p
        if size == 1:
            found[name] = float(source[start])
            continue
        values = source[start : start + size]
        found[name] = values if sites is None else values[sites]
    return found

    BaseSpec() dataclass

    Bases: BaseConfig

    Base of every spec: a frozen dataclass whose fields are what it is and whose check() validates them.

    A spec declares what a class reads, records what a run produced, or holds what a compiler built. It never changes, holds only its fields, check(), and readers of its own fields, and is prepared and run like any configurable object, both doing nothing. It is the only base whose hooks have defaults: a child that validates its fields overrides _check().

    Examples:

    >>> from dataclasses import dataclass
>>> from skcomplex.specs import BaseSpec
>>> @dataclass(frozen=True)
... class SpanSpec(BaseSpec):
...     start: float
...     end: float
>>> SpanSpec(0.0, 1.0).check().end
1.0

    def raster(self, resolution: int, bounds: Sequence[float] | None = None) -> npt.NDArray[np.bool_]:
    """Return a square boolean grid of the cells the geometry covers, with the first row at the top.

    Segments mark the cells of points spaced at most half a cell apart along them, and polygons mark the cells
    whose centres they contain, so the grid suits `ScalingAnalysis('box-counting')`.

    Args:
        resolution:
            The number of cells per side.

        bounds:
            The box `(xmin, ymin, xmax, ymax)` to rasterise, or `None` for the bounding box of the geometry. The
            grid covers a square with the longer side of the box.

    Returns:
        grid:
            The boolean grid of shape `(resolution, resolution)`.

    Raises:
        ValidationError:
            The resolution is not a positive integer.
    """
    if not isinstance(resolution, int | np.integer) or resolution < 1:
        error_msg = f'resolution = {resolution!r} must be a positive integer.'
        raise ValidationError(error_msg)
    xmin, ymin, xmax, ymax = self.bounds() if bounds is None else bounds
    side = max(xmax - xmin, ymax - ymin) or 1.0
    cell = side / resolution
    grid = np.zeros((resolution, resolution), dtype=bool)
    geometry = self.geometry()
    if not len(geometry):
        return grid
    lines = geometry[shapely.get_type_id(geometry) == shapely.GeometryType.LINESTRING]
    if len(lines):
        points = shapely.get_coordinates(shapely.segmentize(lines, cell / 2))
        self._mark(grid, points, xmin, ymax, cell)
    shapes = geometry[shapely.get_type_id(geometry) == shapely.GeometryType.POLYGON]
    if len(shapes):
        centres = (np.arange(resolution) + 0.5) * cell
        x, y = np.meshgrid(xmin + centres, ymax - centres)
        tree = shapely.STRtree(shapes)
        found = tree.query(shapely.points(x.ravel(), y.ravel()), predicate='intersects')[0]
        grid.ravel()[np.unique(found)] = True
    return grid

    def principal_of(self, components: npt.ArrayLike) -> tuple[Array, Array]:
    """Return the principal axis and the scalar order of symmetric tensors of independent components.

    The principal axis is the eigenvector of the largest eigenvalue of the traceless part. The scalar order is that
    eigenvalue times `n_dim / (n_dim - 1)`, so a tensor `S (n n - I / n_dim)` has the order `S`.

    Args:
        components:
            The independent components on the last axis.

    Returns:
        axes:
            The axes, unit vectors on the last axis, whose sign has no meaning.

        orders:
            The scalar orders.

    Raises:
        ValidationError:
            The tensor is not symmetric.

    Examples:
        >>> from skcomplex.specs import TensorSpec
        >>> axes, orders = TensorSpec('traceless', 2).principal_of([0.5, 0.0])
        >>> abs(axes).round(6).tolist(), float(orders)
        ([1.0, 0.0], 1.0)
    """
    if self.symmetry not in {'symmetric', 'traceless'}:
        error_msg = f'A tensor of the symmetry {self.symmetry!r} has no principal axes. Use a symmetric tensor.'
        raise ValidationError(error_msg)
    tensors = self.array_of(components)
    means = np.trace(tensors, axis1=-2, axis2=-1) / self.n_dim
    values, vectors = np.linalg.eigh(tensors - means[..., None, None] * np.eye(self.n_dim))
    return vectors[..., -1], values[..., -1] * self.n_dim / (self.n_dim - 1)

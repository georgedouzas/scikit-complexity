    ModelCatalog(root, package='skcomplex.models')

    Catalog of the models of a package: their listings read from source, and their entries once registered.

    The catalog lists every factory that a module of the package decorates with register_model, reading the registrations from the sources without importing them. Asking for the entry of a listed model imports its module, whose registration adds the entry. Models registered at run time outside the package are listed too.

    def entry(self, name: str) -> ModelEntry:
    """Return a model by name, importing the module of a listed model so that it registers.

    Args:
        name:
            The name of the model.

    Returns:
        entry:
            The registered model.

    Raises:
        RegistryError:
            No model has the name, with the closest names in the message.
    """
    if name not in self._entries:
        listing = self._listed().get(name)
        if listing is not None:
            importlib.import_module(listing.module)
    if name not in self._entries:
        known = sorted({*self._listed(), *self._entries})
        close = difflib.get_close_matches(name, known, n=3)
        hint = f' Did you mean {", ".join(close)}?' if close else ' List the models with `catalog()`.'
        error_msg = f'No model is registered as {name!r}.{hint}'
        raise RegistryError(error_msg)
    return self._entries[name]

    def listings(self, field: str | None = None, tag: str | None = None) -> list[ModelListing]:
    """Return the listed models, optionally of one field or with one tag, sorted by field and name.

    Args:
        field:
            The field, such as `biology`, or the field path, such as `biology/ecology`, to keep, or `None` for
            every field.

        tag:
            The tag to keep, or `None` for every model.

    Returns:
        listings:
            The listed models.

    Raises:
        RegistryError:
            A model registers its name, field, description, tags, or references with a value computed at run time.
    """
    listings = dict(self._listed())
    for name, entry in self._entries.items():
        listings.setdefault(
            name,
            ModelListing(
                entry.name,
                entry.field,
                entry.description,
                entry.tags,
                entry.references,
                entry.factory.__module__,
            ),
        )
    return sorted(
        (
            listing
            for listing in listings.values()
            if (field is None or listing.field == field or listing.field.startswith(f'{field}/'))
            and (tag is None or tag in listing.tags)
        ),
        key=lambda listing: (listing.field, listing.name),
    )

    def register(self, entry: ModelEntry) -> None:
    """Register a model under its name, replacing any previous one.

    An entry without a description takes the first line of its factory's docstring, without a leading `Return`.

    Args:
        entry:
            The registered model.
    """
    if not entry.description:
        entry = dataclasses.replace(entry, description=self._summarize(inspect.getdoc(entry.factory) or ''))
    self._entries[entry.name] = entry

    def check(self) -> 'ModelEntry':
    """Validate that the default run settings are known and every argument of the factory has a default.

    Returns:
        entry:
            The entry itself.

    Raises:
        ValidationError:
            A default run setting is unknown, or a factory argument has no default.
    """
    unknown = set(self.defaults) - set(RUN_SETTINGS)
    if unknown:
        error_msg = f'Unknown run settings {sorted(unknown)} for {self.name}. Use {", ".join(RUN_SETTINGS)}.'
        raise ValidationError(error_msg)
    required = [
        parameter.name
        for parameter in inspect.signature(self.factory).parameters.values()
        if parameter.default is inspect.Parameter.empty
        and parameter.kind not in {inspect.Parameter.VAR_POSITIONAL, inspect.Parameter.VAR_KEYWORD}
    ]
    if required:
        names = ', '.join(required)
        error_msg = f'The factory {self.factory.__name__} needs defaults for {names} to be registered.'
        raise ValidationError(error_msg)
    return self

    def settings_of(self, **parameters: object) -> dict[str, object]:

            A default run setting is unknown, or a factory argument has no default.

    Examples:
        >>> from typing import Annotated
        >>> from skcomplex.core.elements import Element
        >>> from skcomplex.specs import StateSpec
        >>> from skcomplex.core.system import System
        >>> from skcomplex.models import create, register_model
        >>> class PopulationElement(Element):
        ...     def __init__(self, label, n: Annotated[float, StateSpec()] = 1.0):
        ...         super().__init__(label)
        ...         self.n = n
        >>> @register_model(field='examples/demo', references=('This docstring',), defaults={'until': 5})
        ... def constant_population(n: float = 3.0) -> System:
        ...     '''Return one population that does not change.'''
        ...     return System(elements=[PopulationElement('a', n=n)])
        >>> create('constant_population').elements[0].n
        3.0
    """

    def decorator(factory: Factory) -> Factory:
        entry = ModelEntry(
            name or factory.__name__,
            factory,
            field,
            description or '',
            tuple(tags),
            tuple(references),
            dict(defaults or {}),
            observables,
            initial_state,
            settings,
        )
        MODELS.register(entry.check())
        return factory

    return decorator

    def run(name: str, *, settings: Mapping[str, object] | None = None, **parameters: object) -> Simulation:
    """Create a model and simulate it with its default run settings.

    It imports the model's module and nothing else of the library.

    Args:
        name:
            The name of the model.

        settings:
            Run settings that replace the model's defaults: `until` and any `Simulation` argument. The run is seeded
            with `random_state=0` unless the settings give another seed, or `None` for a fresh run.

        **parameters:
            Arguments of the factory that replace its defaults.

    Returns:
        simulation:
            The finished simulation, whose `trajectory_` holds the run.

    Raises:
        RegistryError:
            No model has the name.

        ValidationError:
            The setting `until` is not a number, or `initial_state` is not a mapping.

        ArgumentError:
            A setting is neither `until` nor an argument of `Simulation`.
    """
    entry = MODELS.entry(name)
    chosen: dict[str, object] = {'random_state': 0, **entry.settings_of(**parameters), **(settings or {})}
    until = chosen.pop('until', 10.0)
    if not isinstance(until, int | float):
        error_msg = f'The run setting `until` of {name} is {until!r}. Give it as a number.'
        raise ValidationError(error_msg)
    initial_state = chosen.pop('initial_state', None) or {}
    if not isinstance(initial_state, Mapping):
        error_msg = f'The run setting `initial_state` of {name} is {initial_state!r}. Give it as a mapping.'
        raise ValidationError(error_msg)
    system = entry.factory(**parameters)
    initial = {**entry.initial_state_of(**parameters), **initial_state}
    simulation = Simulation(system, observables=entry.observables_of(system), initial_state=initial)
    unknown = sorted(set(chosen) - set(simulation.get_params(deep=False)))
    if unknown:
        error_msg = f'No run setting named {unknown[0]!r}. Give `until` or an argument of `Simulation`.'
        raise ArgumentError(error_msg)
    return simulation.set_params(**chosen).check().prepare().run(until=float(until))

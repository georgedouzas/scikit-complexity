    ContenderElement(label, visual=VisualSpec('agents', {'color_by': 'strength'}), *, strength=0.0, eta=1.0, relaxation=0.1)

    Bases: Element

    Agent of the Bonabeau model, who wanders on a grid and fights the agents it runs into.

    Every step an agent's strength relaxes by the factor 1 - relaxation, and the agent tries to step into a random neighbouring cell. An attacker beats the occupant with probability 1 / (1 + exp(eta * (h_occupant - h_attacker))), the winner gains 1 and the loser loses 1, and a winning attacker swaps places with the occupant. Agents differ only by their strength, so the swap exchanges their strengths.

    def wander(self, agent: ContenderView, model: ModelView) -> None:
    """Relax the strength, then step into a random neighbouring cell or fight its occupant.

    Args:
        agent:
            The contender.

        model:
            The model.
    """
    agent.strength *= 1 - agent.relaxation
    cells = agent.neighbor_cells()
    cell = cells[int(model.rng.integers(len(cells)))]
    occupants = [
        other
        for other in agent.neighbors()
        if isinstance(other, ContenderView)
        and tuple(np.ravel(np.asarray(other.location))) == tuple(np.ravel(np.asarray(cell)))
    ]
    if occupants:
        self._fight(agent, occupants[0], model)
        return
    agent.move_to(cell)

    def move_free(self, model: ModelView) -> None:
    """Move a random free woman to a random cell of a random city.

    Args:
        model:
            The model.
    """
    free = [woman for woman in model.agents('women', TravellerView) if not woman.victim]
    if not free:
        return
    width, height = self._shape(model)
    city = int(model.rng.integers(self.n_cities))
    start, stop = city * width // self.n_cities, (city + 1) * width // self.n_cities
    cell = (int(model.rng.integers(start, stop)), int(model.rng.integers(height)))
    self._travel(free[int(model.rng.integers(len(free)))], cell, width)

    def move_victims(self, model: ModelView) -> None:
    """Move a random number of the victims of every brothel to random brothels.

    Args:
        model:
            The model.
    """
    women = model.agents('women', TravellerView)
    brothels = [
        DailyMovementsInteraction._cell(place.location)
        for place in model.agents('premises', PremisesView)
        if place.brothel
    ]
    width = self._shape(model)[0]
    for brothel in brothels:
        victims = [
            woman for woman in women if woman.victim and DailyMovementsInteraction._cell(woman.location) == brothel
        ]
        moved = model.rng.permutation(len(victims))[: int(model.rng.integers(len(victims) + 1))]
        for index in moved:
            self._travel(victims[index], brothels[int(model.rng.integers(len(brothels)))], width)

    def execute(self, model: ModelView) -> None:
    """Execute one task and replace it with a new one.

    Args:
        model:
            The model.
    """
    tasks = model.agents(self.target, TaskView)
    if not tasks:
        return
    priorities = [task.priority for task in tasks]
    greedy = model.rng.random() < tasks[0].greed
    chosen = int(np.argmax(priorities)) if greedy else int(model.rng.integers(len(tasks)))
    for index, task in enumerate(tasks):
        task.age += 1
        if index == chosen:
            task.wait, task.age, task.priority = task.age, 0.0, float(model.rng.random())
            model.report('wait', task.wait)

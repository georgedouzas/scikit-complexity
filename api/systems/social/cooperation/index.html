    Bases: Interaction

    Evolutionary dynamics with game transitions, where the game on every edge follows the actions played on it.

    Every edge of the graph carries game 1 or game 2, donation games where a cooperator pays c to give its partner b1 or b2, and every edge starts with game 1. In every time step every player plays the game of every edge with its neighbour and accumulates the payoff pi, of fitness 1 - delta + delta pi. Under death-birth updating a uniformly drawn player dies and its neighbours compete for the site in proportion to their fitness, and under pairwise comparison a uniformly drawn player copies a random neighbour with the probability 1 / (1 + exp(-delta (pi_j - pi_i))). The games then change: an edge goes to game 2 with the probability after_cooperation after mutual cooperation, with after_defection after one defection, and surely after mutual defection, and to game 1 otherwise. The newcomer of a site plays the games of its prior occupant. Every step of the system has one time step per player, and the share of edges with game 1 is reported as game_one.

    def evolve(self, system: SystemView) -> None:
    """Run one time step per player: play the games, replace one player, and change the games.

    Args:
        system:
            The system.
    """
    players = system.agents(self.target, DonorView)
    storage = system.storage
    if 'first' not in storage:
        sources, targets = self._get_edges(players)
        storage['first'], storage['second'] = sources, targets
        storage['games'] = np.ones(sources.size)
        storage['neighbors'] = [
            np.concatenate([targets[sources == node], sources[targets == node]]) for node in range(len(players))
        ]
    first, second, games = storage['first'], storage['second'], storage['games']
    neighbors = storage['neighbors']
    if not isinstance(first, np.ndarray) or not isinstance(second, np.ndarray) or not isinstance(games, np.ndarray):
        return
    if not isinstance(neighbors, list):
        return
    rng = system.rng
    strategies = np.array([player.strategy for player in players])
    payoffs = np.zeros(len(players))
    for _ in range(len(players)):
        payoffs = self._get_payoffs(strategies, games, first, second)
        actions = strategies[first] + strategies[second]
        self._replace(strategies, payoffs, neighbors, rng)
        if self.transitions:
            second_game = np.where(actions == MUTUAL_COOPERATION, self.after_cooperation, self.after_defection)
            second_game = np.where(actions == 0, 1.0, second_game)
            games[:] = np.where(rng.random(games.size) < second_game, 2.0, 1.0)
    for player, strategy, payoff in zip(players, strategies.tolist(), payoffs.tolist(), strict=True):
        player.strategy = strategy
        player.payoff = payoff
    system.report('game_one', float(np.mean(games == 1)) if games.size else 1.0)

def years(
    own: Array,
    other: Array,
    *,
    rounds: float | Array,
    reward: float | Array,
    sucker: float | Array,
    temptation: float | Array,
    punishment: float | Array,
) -> Array:
    """Return the years in prison of a player of strategy `n` against one of strategy `m` in an `N`-round game.

    Args:
        own:
            The strategies `n` of the players.

        other:
            The strategies `m` of their opponents.

        rounds:
            The number `N` of rounds.

        reward:
            The punishment `R` of mutual cooperation.

        sucker:
            The punishment `S` of a betrayed cooperator.

        temptation:
            The punishment `T` of a defector who betrays.

        punishment:
            The punishment `P` of mutual defection.

    Returns:
        years:
            The years in prison of every player.

    Examples:
        >>> import numpy as np
        >>> from skcomplex.systems.social.cooperation import IteratedDilemmaInteraction
        >>> years = IteratedDilemmaInteraction.years(
        ...     np.arange(11.0), np.full(11, 6.0), rounds=10, reward=0.5, sucker=1.5, temptation=0.0, punishment=1.0
        ... )
        >>> int(np.argmin(years))
        5
    """
    first = np.minimum(own, other)
    betrayal = np.where(own < other, temptation, sucker)
    mixed = first * reward + betrayal + (rounds - first - 1) * punishment
    return np.where(own == other, own * reward + (rounds - own) * punishment, mixed)

    def compete(self, system: SystemView) -> None:
    """Draw the births, deaths, and conflicts of one unit of time.

    Args:
        system:
            The system.
    """
    groups = system.agents(self.target, PunisherGroupView)
    size = len(groups)
    if size < PAIR:
        return
    rng = system.rng
    members = float(self.members)
    punishers = np.array([group.punishers for group in groups])
    time = 0.0
    while True:
        punisher_births, defector_births = self._get_birth_rates(punishers)
        rates = np.concatenate([punisher_births, defector_births, np.full(size, float(self.conflict_rate))])
        total = rates.sum()
        time += rng.exponential(1 / total)
        if time > 1.0:
            break
        event = int(np.searchsorted(np.cumsum(rates), rng.random() * total, side='right'))
        group, kind = event % size, event // size
        if kind == 0:
            punishers[group] += rng.random() >= punishers[group] / members
        elif kind == 1:
            punishers[group] -= rng.random() < punishers[group] / members
        else:
            other = int(rng.integers(size - 1))
            other += other >= group
            victory = 0.5 + (punishers[group] - punishers[other]) / (2 * members)
            winner, loser = (group, other) if rng.random() < victory else (other, group)
            punishers[loser] = punishers[winner]
    for group_view, value in zip(groups, punishers.tolist(), strict=True):
        group_view.punishers = value

    cells: int = 100,
    *,
    victory: str = 'punishers',
    group_strength: float = 2.0,
    benefit: float = 2.0,
    cost: float = 1.0,
    punishment: float = 0.5,
    interaction_cost: float = 0.5,
    fixed_cost: float = 0.0,
) -> System:
    """Return the two-level replicator equation of defectors and altruistic punishers, additive pairwise group conflict.

    The density `f(z)` of groups with a fraction `z` of punishers is advected by the replicator dynamics within groups,
    `z (1 - z) (pi_P - pi_D)`, and groups replicate at the rate `group_strength f(z) (G(z) - int G(u) f(u) du)`
    (Cooney 2024, arXiv 2405.18419 Eq. 3.3). Pairwise group conflict with an additively separable victory probability
    `rho(z, u) = 1/2 + (G(z) - G(u)) / 2` reduces to this equation (Eq. 3.11): `G(z) = z` for the probability of
    Boyd and coauthors based on the fraction of punishers (`'punishers'`, Eq. 3.6 and section 5.1), or the average
    payoff of a group normalized by its range, `G(z) / (G* - G_*)` (`'payoff'`, Eq. 3.8 and section 5.2). From a uniform
    density with `G(z) = z` and `p < c + q`, the groups concentrate at the all-defector state below the threshold
    `lambda* = c + q - p`, and above it the mean fraction of punishers tends to `1 - (c + q - p) / lambda` (Eq. 5.4).

    Args:
        cells:
            The number of cells of the axis of the fraction of punishers.

        victory:
            The group-level victory probability, `'punishers'` or `'payoff'`.

        group_strength:
            The relative strength `lambda` of between-group competition.

        benefit:
            The benefit `b` of the donation game.

        cost:
            The cost `c` of cooperating.

        punishment:
            The punishment `p` of a defector by every punisher it meets.

        interaction_cost:
            The cost `k` a punisher pays per defector it meets.

        fixed_cost:
            The fixed cost `q` of being able to punish.

    Returns:
        system:
            The system, with the density `groups.f`.

    Raises:
        ValidationError:
            The victory probability is unknown.

    Examples:
        >>> from skcomplex.systems.social.cooperation import two_level_replicator
        >>> two_level_replicator(40).elements[0].prepare().size_
        40
    """
    _check_victory(victory, ADDITIVE_VICTORIES)
    payoffs = (benefit, cost, punishment, interaction_cost, fixed_cost)
    z, u = coordinate('z'), integration_variable('z')
    gain = _get_group_rate(z, payoffs, victory) - _get_group_rate(u, payoffs, victory)
    advantage = (punishment + interaction_cost) * z - (cost + fixed_cost + interaction_cost)
    space = PhaseSpace(LatticeSpace((1,)), [AxisSpec('z', cells, 0.0, 1.0)])
    groups = FieldElement(
        'groups',
        element=DensityElement,
        space=space,
        values={'f': np.ones((1, cells))},
        discretization='finite-volume',
    )
    within = AdvectionInteraction(
        'within',
        target='groups',
        state='f',
        drift=lambda _: [0, z * (1 - z) * advantage],
        limiter=None,
    )
    between = RateInteraction(
        'between',
        target='groups',
        state='f',
        rate=lambda s: group_strength * s.target.f * integral_transform(s.target.f, gain, ('z',)),
    )
    return System(elements=[groups], interactions=[within, between])

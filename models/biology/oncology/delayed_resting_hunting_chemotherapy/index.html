    Resting and hunting immune cells with a conversion delay under chemotherapy

    The tumour-immune model of resting and hunting cells with a conversion delay, under a drug infusion.

    • Name: delayed_resting_hunting_chemotherapy
    • Field: biology/oncology
    • Formalism: ordinary differential equations
    • References: Padmanabhan, Meskin, and Al Moustafa 2021; Banerjee and Sarkar 2008; Borges et al. 2014

    Factory

    @register_model(
    field='biology/oncology',
    tags=('chemotherapy', 'dde', 'delay', 'tumour-immune'),
    references=(BOOK, 'Banerjee and Sarkar 2008', 'Borges et al. 2014'),
    defaults={'until': 1500.0, 'dt': 5.0},
)
def delayed_resting_hunting_chemotherapy(
    E: float = 0.01,
    Q: float = 0.48,
    A: float = 0.18,
    *,
    u: float = 0.0,
    tau: float = 45.6,
    a: float = 0.001,
) -> System:
    """Return the tumour-immune model of resting and hunting cells with a conversion delay, under a drug infusion.

    Resting immune cells become hunting cells after the delay `tau`, and hunting cells kill the tumour. Time and the
    cells are dimensionless. The parameters and initial conditions are those of Table 3.6 of Padmanabhan et al., and
    the infusion is a constant rate `u`, untreated by default, within the book's range of 0 to 0.03.

    Args:
        E:
            The initial hunting cells.

        Q:
            The initial resting cells, also their history.

        A:
            The initial tumour cells.

        u:
            The constant infusion rate of the drug.

        tau:
            The delay of the conversion of resting cells to hunting cells.

        a:
            The kill rate of the drug on every population.

    Returns:
        system:
            The model, with the cells `patient.E`, `patient.Q`, `patient.A` and the drug `patient.U`.

    Examples:
        >>> from skcomplex.models.biology.oncology import delayed_resting_hunting_chemotherapy
        >>> delayed_resting_hunting_chemotherapy(tau=10.0).elements[0].tau
        10.0
    """
    patient = RestingHuntingChemotherapyElement('patient', E=E, Q=Q, A=A, u=u, tau=tau, a=a)
    drug_effects = [
        interaction
        for name in ('E', 'Q', 'A')
        for interaction in (
            MichaelisMentenInteraction(
                f'drug_kill_{name}',
                target='patient',
                reactants={name: 1},
                substrate=name,
                vmax=lambda s: s.target.a,
                km=lambda s: s.target.bd,
                modifiers={'U': 1},
            ),
            MichaelisMentenInteraction(
                f'drug_uptake_{name}',
                target='patient',
                reactants={'U': 1},
                substrate=name,
                vmax=lambda s: s.target.g,
                km=lambda s: s.target.bd,
                modifiers={'U': 1},
            ),
        )
    ]
    interactions = [
        ReactionInteraction(
            'recruitment',
            target='patient',
            reactants={'Q': 1},
            products={'E': 1},
            modifiers={'E': 1},
            propensity=lambda s: s.target.m1 * s.target.E * delayed(s.target.Q, s.target.tau),
        ),
        ReactionInteraction(
            'effector_death',
            target='patient',
            reactants={'E': 1},
            propensity=lambda s: s.target.d1 * s.target.E,
        ),
        ReactionInteraction(
            'effector_inactivation',
            target='patient',
            reactants={'E': 1},
            modifiers={'A': 1},
            propensity=lambda s: s.target.c1 * s.target.A * s.target.E,
        ),
        RateInteraction(
            'resting_growth',
            target='patient',
            state='Q',
            rate=lambda s: s.target.r1 * s.target.Q * (1 - s.target.b1 * s.target.Q),
        ),
        RateInteraction(
            'tumour_growth',
            target='patient',
            state='A',
            rate=lambda s: s.target.r2 * s.target.A * (1 - s.target.b2 * s.target.A),
        ),
        ReactionInteraction(
            'tumour_kill',
            target='patient',
            reactants={'A': 1},
            modifiers={'E': 1},
            propensity=lambda s: s.target.c2 * s.target.A * s.target.E,
        ),
        DosingInteraction('infusion', target='patient', state='U', dose=lambda s: s.target.u),
        ReactionInteraction(
            'clearance',

class ReactionInteraction(JumpInteraction):
    """Reaction that moves units from its reactants to its products at a mass-action or a given propensity.

    A reactant, a product, or a modifier is a pair of the path of an element, or of a member of a group such as
    `cells[0]`, and the name of its state that holds the units, mapped to its stoichiometric coefficient. A coefficient
    is a positive number, or a function of the namespace of the reaction that returns an expression of its parameters,
    such as `lambda s: 1 / s.target.epsilon`. So `S + I -> 2 I` has the reactants `{('S', 'n'): 1, ('I', 'n'): 1}` and
    the products `{('I', 'n'): 2}`, and two states of one element are `('cell', 'u')` and `('cell', 'v')`. A species may
    also be the name of a state of the `target` alone, such as `{'u': 1, 'v': 2}`. A target that is a group or a field
    reacts in every member, with the species of every member its own states, as the reactions at every site of a
    reaction-diffusion field.

    A subclass may bind its species under roles of its own names, such as `susceptible` and `infected`: it returns its
    entries from `_get_reactants`, `_get_products`, and `_get_modifiers`, pairs of the path a role binds and a state
    name, and the reaction reads every pair from the namespace of the role that binds its path.

    The propensity is `k` times the product of every reactant and modifier state to the power of its coefficient, or the
    expression the `propensity` function returns. The reaction changes every state by its coefficient among the products
    minus its coefficient among the reactants, so it conserves the total of the states whose changes sum to zero, and
    modifiers enter the propensity but do not change.

    In the mode `'rate'` the reaction is a rate equation. In the mode `'jump'` it is the random event of a
    `JumpInteraction` at the propensity that moves the units of the coefficients. In the mode `'langevin'` it is the
    chemical Langevin equation, the rate equation with one Wiener process of amplitude the square root of the propensity
    driving every state the reaction changes.

    In the `'agent'` form the target is an agent group and every member reacts on its own states, the species named by
    state names alone, in every step from the states at its start. The propensity of a member is `k` times the product
    of its reactant and modifier states to the power of their coefficients, or the number `propensity(agent, model)`
    returns, and a coefficient that is a function is the number `coefficient(agent, model)` returns. In the mode
    `'rate'` the member adds the changes times its propensity, one unit step of the rate equation, and in the mode
    `'jump'` it adds the changes with the probability `1 - exp(-propensity)` that an event fires in the step. The
    members react at once from the states at the start of the rule under the schedule `'synchronous'`, one after the
    other in their order or in a new random order every step under `'sequential'` or `'random-sequential'`, or each in
    its own activation under `'member'`.

    Args:
        label:
            The label of the reaction, unique among the parts of its system.

        target:
            The path of the element, group, or field whose states the species named by a state alone are, or `None`.

        reactants:
            The coefficients of the reactants, or `None` for a source.

        products:
            The coefficients of the products, or `None` for a sink.

        k:
            The rate constant.

        propensity:
            A function of the namespace of the reaction that returns the propensity, or `None` for mass action. The
            namespace holds the tuples `reactants`, `products`, and `modifiers` of the namespaces of the elements their
            pairs bind, in the order of their entries, `target`, the namespace of the target or of one of its members,
            and the symbol `k`. In the agent form, a function `propensity(agent, model)` that returns a number.

        modifiers:
            The coefficients of the modifiers, whose states the propensity reads but the reaction does not change, or
            `None`.

        mode:
            The form of the laws, `'rate'`, `'jump'`, or `'langevin'`.

        form:
            `'symbolic'` for the laws of the states, or `'agent'` for a rule in which every member of an agent group
            reacts on its own states.

        phase:
            The phase of the step the agent form runs in, or `None` for the default phase.

        schedule:
            The order of the members in the agent form, `'synchronous'`, `'sequential'`, `'random-sequential'`, or
            `'member'`.

    Raises:
        ValidationError:
            The mode, the form, or the schedule is unknown, the symbolic form is not synchronous, the reaction has
            neither reactants nor products, an entry is neither a pair of a path and a state name nor a state name of a
            given target, a coefficient is neither a positive number nor a function, or the agent form has no target,
            has an entry that is a pair, or has the mode `'langevin'`.

        self,
        label: str,
        *,

        reactants: Annotated[Stoichiometry | None, RoleSpec(variadic=True)] = None,
        products: Annotated[Stoichiometry | None, RoleSpec(variadic=True)] = None,

        modifiers: Annotated[Stoichiometry | None, RoleSpec(variadic=True)] = None,
        mode: str = 'rate',
        form: str = 'symbolic',
        phase: str | None = None,

    def _get_reactants(self) -> Stoichiometry | None:
        """Return the coefficients of the reactants, which a subclass with roles of its own names returns."""
        return self.reactants

    def _get_products(self) -> Stoichiometry | None:
        """Return the coefficients of the products, which a subclass with roles of its own names returns."""
        return self.products

    def _get_modifiers(self) -> Stoichiometry | None:
        """Return the coefficients of the modifiers, which a subclass with roles of its own names returns."""
        return self.modifiers

    def _get_mode(self) -> str:
        """Return the form of the laws, which a subclass that sets it by another argument returns."""
        return self.mode

    def _get_bound(self, namespace: SimpleNamespace) -> dict[str, SimpleNamespace]:
        """Return the namespace of every path the plain roles of the reaction bind, by path."""
        roles = [
            name
            for name, spec in self._get_specs().items()
            if isinstance(spec, RoleSpec) and not spec.variadic and not spec.noise
        ]
        return {
            getattr(self, name): getattr(namespace, name)
            for name in roles
            if isinstance(getattr(self, name), str) and hasattr(namespace, name)
        }

    def _get_terms(
        self,
        entries: Stoichiometry | None,
        namespace: SimpleNamespace,
        name: str,
    ) -> list[tuple[Expression, Expression | float]]:
        """Return the state symbol and the coefficient of every entry of a kind of species.

        A state name alone is a state of the target. A pair reads the namespace of the plain role that binds its path,
        or else the namespace of its variadic role `name`, in the order of the pairs. A coefficient that is a function
        is its expression at the namespace.
        """
        entries = entries or {}
        pairs = [entry for entry in entries if not isinstance(entry, str)]
        sources: dict[object, SimpleNamespace] = {}
        if pairs and name in self._get_specs():
            sources = dict(zip(pairs, getattr(namespace, name), strict=True))
        elif pairs:
            bound = self._get_bound(namespace)
            sources = {pair: bound[pair[0]] for pair in pairs}
        return [
            (
                getattr(
                    self._get_member(namespace, 'target') if isinstance(entry, str) else sources[entry],
                    self._get_state(entry),
                ),
                SR(coefficient(namespace)) if callable(coefficient) else coefficient,
            )
            for entry, coefficient in entries.items()
        ]

    def _get_state(self, entry: tuple[str, str] | str) -> str:
        """Return the name of the state of an entry, a state name alone or the second item of a pair."""
        return entry if isinstance(entry, str) else entry[1]

    def _get_changes(self, namespace: SimpleNamespace) -> dict[Expression, float]:
        """Return the net change of every state the reaction changes, products minus reactants."""
        changes: dict[Expression, float] = {}
        for sign, entries, name in (
            (-1, self._get_reactants(), 'reactants'),
            (1, self._get_products(), 'products'),
        ):
            for state, coefficient in self._get_terms(entries, namespace, name):
                changes[state] = changes.get(state, 0) + sign * coefficient
        return {state: change for state, change in changes.items() if change}

    def _get_mass_action(self, namespace: SimpleNamespace, constant: Expression) -> Expression:
        """Return the mass action of a rate constant, times every reactant and modifier to its coefficient."""
        propensity = SR(constant)
        for entries, name in ((self._get_reactants(), 'reactants'), (self._get_modifiers(), 'modifiers')):
            for state, coefficient in self._get_terms(entries, namespace, name):
                propensity *= state**coefficient
        return propensity

    def _get_rate(self, namespace: SimpleNamespace) -> Expression:
        """Return the propensity of the reaction, by the user's function or by mass action at the rate constant `k`."""
        if self.propensity is not None:
            return SR(self.propensity(namespace))
        return self._get_mass_action(namespace, self._get_parameter_symbol('k', namespace))

    def _get_langevin_laws(self, propensity: Expression, changes: dict[Expression, float]) -> list[BasePart]:
        """Return the drift of the reaction and the noise of its one Wiener process on every state it changes."""
        amplitude = sqrt(where(propensity > 0, propensity, 0))
        rates: list[BasePart] = [RateLaw(state, change * propensity) for state, change in changes.items()]
        noises: list[BasePart] = [
            NoiseLaw(state, change * amplitude, process=self.label) for state, change in changes.items()
        ]
        return rates + noises

    def _is_entry(self, entry: object) -> bool:
        """Check whether an entry is a pair of a path and a state name, or a state name of a given target."""
        if isinstance(entry, str):
            return self.target is not None and entry.isidentifier()
        return (
            isinstance(entry, tuple)
            and len(entry) == ENTRY_SIZE
            and all(isinstance(item, str) for item in entry)
            and entry[1].isidentifier()
        )

    def _is_coefficient(self, coefficient: object) -> bool:
        """Check whether a coefficient is a positive number or a function."""
        return callable(coefficient) or (isinstance(coefficient, int | float) and coefficient > 0)

    def _check_entries(self, name: str, entries: Stoichiometry | None) -> None:
        """Raise an error for entries that are neither pairs nor state names of the target, or not coefficients."""
        if entries is None:
            return
        valid = isinstance(entries, Mapping) and all(
            self._is_entry(entry) and self._is_coefficient(coefficient) for entry, coefficient in entries.items()
        )
        if not valid:
            error_msg = (
                f'The {name} of reaction {self.label!r} are {entries!r}. Map pairs of a path and a state name, or '
                'state names of the target, to positive numbers or functions, such as {("S", "n"): 1} or {"n": 1}.'
            )
            raise ValidationError(error_msg)

    def _check_form(self) -> None:
        """Raise an error for an unknown form, or an agent form without a target, with pairs, or of Langevin mode."""
        if self.form not in FORMS:
            error_msg = f'Reaction {self.label!r} has the form {self.form!r}. Use one of {", ".join(FORMS)}.'
            raise ValidationError(error_msg)
        self._check_variant('schedule', self.schedule, SCHEDULES)
        if self.form != 'agent' and self.schedule != 'synchronous':
            error_msg = f'The schedule {self.schedule!r} of reaction {self.label!r} runs agent rules. Use form="agent".'
            raise ValidationError(error_msg)
        if self.form != 'agent':
            return
        if self.target is None:
            error_msg = f'Reaction {self.label!r} has the agent form but no target. Give the agent group as `target`.'
            raise ValidationError(error_msg)
        mode = self._get_mode()
        if mode not in AGENT_MODES:
            error_msg = (
                f'Reaction {self.label!r} has the agent form and the mode {mode!r}. Use {" or ".join(AGENT_MODES)}.'
            )
            raise ValidationError(error_msg)
        entries = [*(self._get_reactants() or {}), *(self._get_products() or {}), *(self._get_modifiers() or {})]
        if not all(isinstance(entry, str) for entry in entries):
            error_msg = (
                f'Reaction {self.label!r} has the agent form, whose species are states of every member. Name them by '
                'state names alone, such as {"n": 1}.'
            )
            raise ValidationError(error_msg)

    def _check_jump(self) -> None:
        """Raise an error for an unknown mode or form, no reactants and products, or entries the reaction cannot take.

        Raises:
            ValidationError:
                The mode or the form is unknown, the reaction has neither reactants nor products, an entry is neither a
                pair of a path and a state name nor a state name of a given target, a coefficient is neither a positive
                number nor a function, or the agent form has no target, has an entry that is a pair, or has the mode
                `'langevin'`.
        """
        mode = self._get_mode()
        if mode not in MODES:
            error_msg = f'Reaction {self.label!r} has the mode {mode!r}. Use one of {", ".join(MODES)}.'
            raise ValidationError(error_msg)
        reactants, products = self._get_reactants(), self._get_products()
        if not reactants and not products:
            error_msg = f'Reaction {self.label!r} has neither reactants nor products. Give at least one of them.'
            raise ValidationError(error_msg)
        for name, entries in (('reactants', reactants), ('products', products), ('modifiers', self._get_modifiers())):
            self._check_entries(name, entries)
        self._check_form()

    def _get_number(self, value: object, agent: AgentViewProtocol, model: ModelViewProtocol) -> float:
        """Return a number, or the value of a function of an agent and the model as a number."""
        found = value(agent, model) if callable(value) else value
        if not isinstance(found, SupportsFloat):
            error_msg = f'A function of reaction {self.label!r} returns {found!r} for an agent. Return a number.'
            raise ValidationError(error_msg)
        return float(found)

    def _get_agent_changes(self, agent: AgentViewProtocol, model: ModelViewProtocol) -> dict[str, float]:
        """Return the net change of every state of an agent the reaction changes, products minus reactants."""
        changes: dict[str, float] = {}
        for sign, entries in ((-1, self._get_reactants()), (1, self._get_products())):
            for entry, coefficient in (entries or {}).items():
                state = self._get_state(entry)
                changes[state] = changes.get(state, 0.0) + sign * self._get_number(coefficient, agent, model)
        return changes

    def _get_agent_propensity(self, agent: AgentViewProtocol, model: ModelViewProtocol) -> float:
        """Return the propensity of an agent, by the user's function or by mass action at the rate constant `k`."""
        if self.propensity is not None:
            return self._get_number(self.propensity, agent, model)
        propensity = self._get_parameter_value('k', model)
        for entries in (self._get_reactants(), self._get_modifiers()):
            for entry, coefficient in (entries or {}).items():
                value = float(getattr(agent, self._get_state(entry)))
                propensity *= value ** self._get_number(coefficient, agent, model)
        return propensity

    def _get_agent_scale(self, propensity: float, model: ModelViewProtocol) -> float:
        """Return the multiple of its changes an agent adds in a step, its propensity or whether an event fires."""
        if self._get_mode() == 'rate':
            return propensity
        return float(model.rng.random() < 1.0 - math.exp(-max(propensity, 0.0)))

    def _get_agent_step(self, agent: AgentViewProtocol, model: ModelViewProtocol) -> dict[str, float]:
        """Return the change of every state of an agent in the step, its changes times its propensity or its event."""
        changes, propensity = self._get_agent_changes(agent, model), self._get_agent_propensity(agent, model)
        scale = self._get_agent_scale(propensity, model)
        return {state: change * scale for state, change in changes.items()}

    def _apply_agent_step(self, agent: AgentViewProtocol, changes: dict[str, float]) -> None:
        """Add their changes to the states of an agent."""
        for state, change in changes.items():
            setattr(agent, state, float(getattr(agent, state)) + change)

    def _react_member(self, agent: AgentViewProtocol, model: ModelViewProtocol) -> None:
        """React in one agent on its own states, in its own activation."""
        if agent.alive:
            self._apply_agent_step(agent, self._get_agent_step(agent, model))

    def react(self, model: ModelViewProtocol) -> None:
        """React in every agent of the target on its own states, by the schedule.

        Args:
            model:
                The model view of the interaction.
        """
        self._update_agents(model, self.schedule, self._get_agent_step, self._apply_agent_step)

    def _get_laws(self, namespace: SimpleNamespace) -> list[BasePart]:
        """Return the laws of the reaction in its mode, or the rule of the agent form.

        Args:
            namespace:
                The namespace of the symbols the laws can use, with the namespaces of the target, the reactants, the
                products, and the modifiers.

        Returns:
            laws:
                The rate laws, the jump law, the rate and noise laws, or the rule of the reaction.
        """
        if self.form == 'agent' and self.schedule == 'member':
            return [RuleLaw(self._react_member, phase=self.phase, members=self._get_role('target'))]
        if self.form == 'agent':
            return [RuleLaw(self.react, phase=self.phase)]
        mode = self._get_mode()
        if mode == 'jump':
            return [*super()._get_laws(namespace)]
        propensity = self._get_rate(namespace)
        changes = self._get_changes(namespace)
        if mode == 'langevin':
            return self._get_langevin_laws(propensity, changes)
        return [RateLaw(state, change * propensity) for state, change in changes.items()]

class RateInteraction(Interaction):
    """Rate `rate(ns)` and noise `noise(ns) dW` added to the highest time derivative of a state of a target.

    On a state of order 1 the rate is its first derivative, and on a state of order `k` it adds to the `k`-th
    derivative, so rates on one state superpose. With a divisor `d` the rate and the noise are divided by `d`, the name
    of a state or parameter of the target, such as an inertia, a friction, or a capacitance, which backends can read.

    A rate, a noise, or a divisor function takes the namespace of the interaction, with the namespaces `target` and
    `sources` of the bound elements, the time `t`, and the symbols of the parameters a subclass declares. A number
    stands for a constant, and a number on a vector state applies to every component. The noise is a Wiener process per
    component and member by default, a process of a `WienerProcessSpec` to share or correlate it across states and
    members, or a fractional Brownian motion of a Hurst index.

    Args:
        label:
            The label of the interaction, unique among the parts of its system.

        target:
            The path of the element or group whose state the rate changes.

        state:
            The name of the state of the target that the rate changes.

        rate:
            A function of the namespace that returns the rate, or a number.

        sources:
            The paths of the elements or groups the rate and the noise read but do not change.

        noise:
            A function of the namespace that returns the amplitude of the noise, a number, or `None` for no noise.

        divisor:
            The name of the state or parameter of the target that divides the rate and the noise, a function of the
            namespace of the target that returns the divisor, such as a moment of inertia `m l^2`, or `None`.

        process:
            The Wiener processes that drive the noise, or `None` for one independent process per component and member.

        index:
            The process of `process` that drives the noise, or `None` for every process, one per component of a vector
            state.

        convention:
            The stochastic calculus of the noise, `'ito'`, `'stratonovich'`, or `'anti-ito'`.

        hurst:
            The Hurst index in `(0, 1)` of a fractional Brownian motion that drives the noise, a number or a function
            of the namespace that returns a parameter, or `None` for a Wiener process.

        conserved:
            Whether the noise is the divergence of a random flux on the links of the lattice of a field, so it keeps the
            total of the state.

    Raises:
        ValidationError:
            The state is not a name, a function is neither callable nor a number, the convention is unknown, the Hurst
            index is outside `(0, 1)`, or the noise options cannot combine.

    Examples:
        >>> from skcomplex.core.elements import CompartmentElement
        >>> from skcomplex.core.interactions import RateInteraction
        >>> from skcomplex.core.system import System
        >>> from skcomplex.engine.runtime import Simulation
        >>> decay = RateInteraction('decay', target='stock', state='n', rate=lambda s: -0.5 * s.target.n)
        >>> system = System(elements=[CompartmentElement('stock', n=8.0)], interactions=[decay])
        >>> trajectory = Simulation(system, dt=0.01).check().prepare().run(until=2.0).trajectory_
        >>> round(float(trajectory.values('stock.n')[-1]), 3)
        2.943
    """

    def __init__(
        self,
        label: str,
        *,
        target: Annotated[str, RoleSpec()],
        state: str = 'x',
        rate: Callable[[SimpleNamespace], object] | float = 0.0,
        sources: Annotated[Sequence[str], RoleSpec(variadic=True)] = (),
        noise: Callable[[SimpleNamespace], object] | float | None = None,
        divisor: str | Callable[[SimpleNamespace], object] | None = None,
        process: Annotated[WienerProcessSpec | None, RoleSpec(noise=True)] = None,
        index: int | None = None,
        convention: str = 'ito',
        hurst: Callable[[SimpleNamespace], object] | float | None = None,
        conserved: bool = False,
    ) -> None:
        super().__init__(label)
        self.target = target
        self.state = state
        self.rate = rate
        self.sources = sources
        self.noise = noise
        self.divisor = divisor
        self.process = process
        self.index = index
        self.convention = convention
        self.hurst = hurst
        self.conserved = conserved

    def _check_functions(self) -> None:
        """Raise an error for a state that is not a name, or a function that is neither callable nor a number."""
        if not isinstance(self.state, str) or not self.state.isidentifier():
            error_msg = (
                f'{type(self).__name__} {self.label!r} changes the state {self.state!r}, which is not a name of a '
                'state.'
            )
            raise ValidationError(error_msg)
        for name in ('rate', 'noise'):
            value = getattr(self, name)
            if not (value is None or callable(value) or isinstance(value, int | float)):
                error_msg = f'The {name} of {self.label!r} is {value!r}. Give a function of the namespace or a number.'
                raise ValidationError(error_msg)
        if not (
            self.divisor is None
            or callable(self.divisor)
            or (isinstance(self.divisor, str) and self.divisor.isidentifier())
        ):
            error_msg = (
                f'The divisor of {self.label!r} is {self.divisor!r}. Give the name of a state or parameter, or a '
                'function of the namespace of the target.'
            )
            raise ValidationError(error_msg)

    def _check_noise(self) -> None:
        """Raise an error for an unknown convention, a Hurst index outside `(0, 1)`, or noise options that conflict."""
        if self.convention not in CONVENTIONS:
            error_msg = f'Unknown noise convention {self.convention!r}. Use one of {", ".join(CONVENTIONS)}.'
            raise ValidationError(error_msg)
        if not (
            self.hurst is None or callable(self.hurst) or (isinstance(self.hurst, int | float) and 0 < self.hurst < 1)
        ):
            error_msg = f'The Hurst index of {self.label!r} is {self.hurst!r}. Give a number in (0, 1) or a function.'
            raise ValidationError(error_msg)
        if self.index is not None and self.process is None:
            error_msg = f'{self.label!r} picks the process {self.index!r} but binds no processes. Give `process`.'
            raise ValidationError(error_msg)
        options = [
            name
            for name, given in (('hurst', self.hurst), ('process', self.process), ('conserved', self.conserved))
            if given
        ]
        if len(options) > 1:
            error_msg = f'The noise of {self.label!r} takes {" and ".join(options)}. Give at most one of them.'
            raise ValidationError(error_msg)

    def _check(self) -> None:
        """Raise an error for a state, functions, or noise options the interaction cannot take.

        Raises:
            ValidationError:
                The type declares states, the state is not a name, a function is neither callable nor a number, the
                convention is unknown, the Hurst index is outside `(0, 1)`, or the noise options cannot combine.
        """
        super()._check()
        self._check_functions()
        self._check_noise()

    def _evaluate(self, function: Callable[[SimpleNamespace], object] | float, namespace: SimpleNamespace) -> object:
        """Return the value of a function of the namespace, or a number as it is."""
        return function(namespace) if callable(function) else function

    def _broadcast(self, value: object, state: object) -> object:
        """Return a value with one component per component of a vector state, repeating a number."""
        if isinstance(state, FreeModuleElement) and not isinstance(value, FreeModuleElement | list | tuple):
            return vector(SR, [value] * len(state))
        if isinstance(value, list | tuple):
            return vector(SR, value)
        return value

    def _divide(self, value: object, namespace: SimpleNamespace) -> object:
        """Return a value divided by the divisor of the target, or the value without a divisor."""
        if self.divisor is None:
            return value
        divisor = (
            self.divisor(self._get_member(namespace, 'target'))
            if callable(self.divisor)
            else getattr(self._get_member(namespace, 'target'), self.divisor)
        )
        return operator.truediv(value, divisor)

    def _get_rate(self, namespace: SimpleNamespace) -> object:
        """Return the rate before the divisor, the hook a subclass overrides to derive it from its parameters."""
        return self._evaluate(self.rate, namespace)

    def _get_noise(self, namespace: SimpleNamespace) -> object:
        """Return the noise amplitude before the divisor, the hook a subclass overrides, or `None` for no noise."""
        return None if self.noise is None else self._evaluate(self.noise, namespace)

    def _get_noise_law(self, state: object, amplitude: object, namespace: SimpleNamespace) -> BasePart:
        """Return the noise law of an amplitude, by the process, the convention, and the Hurst index."""
        if self.hurst is not None:
            hurst = self._evaluate(self.hurst, namespace)
            return FractionalNoiseLaw(state, amplitude, hurst=hurst, convention=self.convention)
        process = None
        if self.process is not None:
            handle = self._get_member(namespace, 'process').dW
            process = handle if self.index is None else handle[self.index]
        return NoiseLaw(state, amplitude, convention=self.convention, process=process, conserved=self.conserved)

    def _get_laws(self, namespace: SimpleNamespace) -> list[BasePart]:
        """Return the rate law of the state and its noise law, both divided by the divisor.

        Args:
            namespace:
                The namespace of the symbols the laws can use, with the namespaces of the target and the sources.

        Returns:
            laws:
                The rate law, and the noise law when the interaction has noise.
        """
        state = getattr(self._get_member(namespace, 'target'), self.state)
        rate = self._divide(self._broadcast(self._get_rate(namespace), state), namespace)
        laws: list[BasePart] = [RateLaw(state, rate)]
        noise = self._get_noise(namespace)
        if noise is not None:
            laws.append(self._get_noise_law(state, self._divide(noise, namespace), namespace))
        return laws

class DosingInteraction(RateInteraction):
    """Input of a drug into a state of a target at the rate `dose`, constantly or in a schedule of infusions.

    Without a period the drug enters at the constant rate `dose`. With a period it enters at the rate `dose` for `width`
    time units every `period` from `start`, `count` times or without end. The dose is a function of the namespace,
    which reads the parameters of the target, or a number.

    Args:
        label:
            The label of the interaction, unique among the parts of its system.

        target:
            The path of the element or group whose drug state the dose raises.

        state:
            The name of the state of the target that holds the drug.

        dose:
            A function of the namespace that returns the rate of the input during an infusion, or a number.

        period:
            The time between the starts of infusions, or `None` for a constant input.

        width:
            The duration of every infusion.

        start:
            The time of the first infusion.

        count:
            The number of infusions, or `None` for a schedule that never ends.

    Raises:
        ValidationError:
            The state is not a name, the dose is neither a function nor a number, or the schedule is not positive.

    Examples:
        >>> from skcomplex.core.elements import CompartmentElement
        >>> from skcomplex.core.system import System
        >>> from skcomplex.engine.runtime import Simulation
        >>> from skcomplex.models.biology.common.pharmacokinetics import DosingInteraction
        >>> dosing = DosingInteraction('dosing', target='patient', state='n', dose=2.0, period=10.0, count=3)
        >>> system = System(elements=[CompartmentElement('patient', n=0.0)], interactions=[dosing])
        >>> trajectory = Simulation(system, dt=0.5).check().prepare().run(until=40.0).trajectory_
        >>> round(float(trajectory.values('patient.n')[-1]), 3)
        6.0
    """

    def __init__(
        self,
        label: str,
        *,
        target: Annotated[str, RoleSpec()],
        state: str = 'x',
        dose: Callable[[SimpleNamespace], object] | float = 1.0,
        period: float | None = None,
        width: float = 1.0,
        start: float = 0.0,
        count: int | None = None,
    ) -> None:
        super().__init__(label, target=target, state=state)
        self.dose = dose
        self.period = period
        self.width = width
        self.start = start
        self.count = count

    def _check(self) -> None:
        """Raise an error for a state, a dose, or a schedule the input cannot take.

        Raises:
            ValidationError:
                The state is not a name, the dose is neither a function nor a number, or the schedule is not positive.
        """
        super()._check()
        if not (callable(self.dose) or isinstance(self.dose, int | float)):
            error_msg = f'The dose of {self.label!r} is {self.dose!r}. Give a function of the namespace or a number.'
            raise ValidationError(error_msg)
        if self.period is not None and not (self.period > 0 and self.width > 0):
            error_msg = (
                f'The schedule of {self.label!r} has the period {self.period!r} and the width {self.width!r}. Give '
                'both above 0.'
            )
            raise ValidationError(error_msg)

    def _get_rate(self, namespace: SimpleNamespace) -> Expression:
        """Return the dose, times the schedule of infusions when the input has a period.

        Args:
            namespace:
                The namespace of the interaction, with the namespace of the target and the time `t`.

        Returns:
            rate:
                The rate of the input.
        """
        dose = self._evaluate(self.dose, namespace)
        if self.period is None:
            constant: Expression = dose
            return constant
        schedule: Expression = dose * pulse_train(
            namespace.t,
            self.period,
            self.width,
            start=self.start,
            count=self.count,
        )
        return schedule

class MichaelisMentenInteraction(ReactionInteraction):
    """Reaction at the Michaelis-Menten propensity `vmax S / (km + S)`, times every modifier to its coefficient.

    The substrate `S` is a state of the target, or a function of the namespace that returns an amount that is no state
    of its own, such as the phosphorylated form `A_T - A` of a protein whose total is conserved. The modifiers are the
    enzymes that catalyse the reaction, which scale the propensity but do not change, so `vmax` is a catalytic rate when
    the reaction has modifiers and a maximal rate when it has none. The maximal rate and the Michaelis constant are
    functions of the namespace, which reads the parameters of the target, or numbers.

    Args:
        label:
            The label of the reaction, unique among the parts of its system.

        target:
            The path of the element, group, or field whose states the species named by a state alone are.

        reactants:
            The coefficients of the reactants, or `None` for a source.

        products:
            The coefficients of the products, or `None` for a sink.

        substrate:
            The name of the state of the target that saturates the propensity, or a function of the namespace that
            returns the amount that does.

        vmax:
            A function of the namespace that returns the maximal or catalytic rate, or a number.

        km:
            A function of the namespace that returns the Michaelis constant, or a number.

        modifiers:
            The coefficients of the modifiers, the enzymes that scale the propensity, or `None`.

        mode:
            The form of the laws, `'rate'`, `'jump'`, or `'langevin'`.

    Raises:
        ValidationError:
            The substrate is neither a name nor a function, a rate is neither a function nor a number, or the reaction
            cannot take its species or mode.

    Examples:
        >>> from skcomplex.core.elements import CompartmentElement
        >>> from skcomplex.core.system import System
        >>> from skcomplex.engine.runtime import Simulation
        >>> from skcomplex.models.biology.common.kinetics import MichaelisMentenInteraction
        >>> uptake = MichaelisMentenInteraction(
        ...     'uptake', target='cell', reactants={'n': 1}, substrate='n', vmax=1.0, km=100.0
        ... )
        >>> system = System(elements=[CompartmentElement('cell', n=500.0)], interactions=[uptake])
        >>> trajectory = Simulation(system, dt=1.0).check().prepare().run(until=10.0).trajectory_
        >>> round(float(trajectory.values('cell.n')[-1]), 3)
        491.679
    """

    def __init__(
        self,
        label: str,
        *,
        target: Annotated[str | None, RoleSpec()] = None,
        reactants: Annotated[Stoichiometry | None, RoleSpec(variadic=True)] = None,
        products: Annotated[Stoichiometry | None, RoleSpec(variadic=True)] = None,
        substrate: str | Callable[[SimpleNamespace], object] = 'n',
        vmax: Callable[[SimpleNamespace], object] | float = 1.0,
        km: Callable[[SimpleNamespace], object] | float = 1.0,
        modifiers: Annotated[Stoichiometry | None, RoleSpec(variadic=True)] = None,
        mode: str = 'rate',
    ) -> None:
        super().__init__(label, target=target, reactants=reactants, products=products, modifiers=modifiers, mode=mode)
        self.substrate = substrate
        self.vmax = vmax
        self.km = km

    def _check(self) -> None:
        """Raise an error for a substrate or rates the reaction cannot take.

        Raises:
            ValidationError:
                The substrate is neither a name nor a function, a rate is neither a function nor a number, or the
                reaction cannot take its species or mode.
        """
        super()._check()
        if not (callable(self.substrate) or (isinstance(self.substrate, str) and self.substrate.isidentifier())):
            error_msg = (
                f'The substrate of {self.label!r} is {self.substrate!r}. Give the name of a state of the target or a '
                'function of the namespace.'
            )
            raise ValidationError(error_msg)
        for name in ('vmax', 'km'):
            value = getattr(self, name)
            if not (callable(value) or isinstance(value, int | float)):
                error_msg = f'The {name} of {self.label!r} is {value!r}. Give a function of the namespace or a number.'
                raise ValidationError(error_msg)

    def _evaluate(self, value: Callable[[SimpleNamespace], object] | float, namespace: SimpleNamespace) -> object:
        """Return the value of a function of the namespace, or a number as it is."""
        return value(namespace) if callable(value) else value

    def _get_rate(self, namespace: SimpleNamespace) -> Expression:
        """Return the saturating propensity of the substrate, times every modifier to its coefficient.

        Args:
            namespace:
                The namespace of the reaction, with the namespace of the target.

        Returns:
            propensity:
                The propensity of the reaction.
        """
        if callable(self.substrate):
            substrate = SR(self.substrate(namespace))
        else:
            substrate = SR(getattr(namespace.target, self.substrate))
        vmax, km = SR(self._evaluate(self.vmax, namespace)), SR(self._evaluate(self.km, namespace))
        propensity = vmax * substrate / (km + substrate)
        for state, coefficient in self._get_terms(self._get_modifiers(), namespace, 'modifiers'):
            propensity *= state**coefficient
        return propensity
